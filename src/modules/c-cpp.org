#+TITLE: C/C++ settings
#+AUTHOR: Diacus Magnuz
#+STARTUP: overview

Most of the IDE configuration for C/C++ is taken from [[https://emacs-lsp.github.io/lsp-mode/tutorials/CPP-guide/][this guide]].

#+begin_src emacs-lisp
  (add-hook 'c-mode-hook #'lsp)
  (add-hook 'c++-mode-hook #'lsp)

  ;; Main settings for clangd
  (with-eval-after-load 'lsp-mode
    (setq lsp-clients-clangd-args
	  '("--background-index"
	    "--compile-commands-dir=."  ;; Search in current directory
	    "--header-insertion=never"
	    "--clang-tidy"
	    "--all-scopes-completion"
	    "--completion-style=detailed"
	    "--pch-storage=memory"
	    "--fallback-style=llvm"
	    "--header-insertion-decorators=0")))
#+end_src
