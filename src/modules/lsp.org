#+TITLE: Eglot LSP Settings
#+AUTHOR: Diacus Magnuz

[[https://www.gnu.org/software/emacs/manual/html_node/eglot/][Eglot]] is a lightweight LSP client built into Emacs 29+. It provides
essential Language Server Protocol functionality with minimal configuration
compared to the more feature-heavy lsp-mode.

* Performance Tuning
Settings to improve responsiveness (analogous to lsp-mode settings).

#+begin_src emacs-lisp
  (setq gc-cons-threshold (* 100 1024 1024))
  (setq read-process-output-max (* 1024 1024))
#+end_src

* Core Configuration
#+begin_src emacs-lisp
  (use-package eglot
    :ensure nil  ; Built into Emacs 29+
    :config
    (setq eglot-sync-connect 1)              ; Sync connect timeout (seconds)
    (setq eglot-autoshutdown t)               ; Shutdown server when last buffer closes
    (setq eglot-send-changes-idle-time 0.1))  ; Send changes quickly
#+end_src

* Company Integration
Eglot uses company-capf (completion-at-point) backend.

#+begin_src emacs-lisp
  (setq company-idle-delay 0.0)
  (setq company-minimum-prefix-length 1)

  (with-eval-after-load 'company
    (cl-pushnew 'company-capf company-backends))
#+end_src

* Keybindings
#+begin_src emacs-lisp
  (defun eglot--key-bindings ()
    "Define key bindings for eglot."
    (evil-define-key '(normal visual motion) 'global
      (kbd "SPC l r") 'eglot-rename
      (kbd "SPC l f") 'eglot-format
      (kbd "SPC l a") 'eglot-code-actions
      (kbd "SPC l F") 'eglot-format-buffer))

  (add-hook 'eglot-managed-mode-hook 'eglot--key-bindings)
#+end_src

* Language Server Configuration
Configure language servers that require custom arguments.

#+begin_src emacs-lisp
  (with-eval-after-load 'eglot
    ;; Clangd for C/C++
    (add-to-list 'eglot-server-programs
                 '((c-mode c++-mode)
                   . ("clangd"
                      "--background-index"
                      "--compile-commands-dir=."
                      "--header-insertion=never"
                      "--clang-tidy"
                      "--all-scopes-completion"
                      "--completion-style=detailed"
                      "--pch-storage=memory"
                      "--fallback-style=llvm"))))
#+end_src

* Notes
- Language servers must be installed system-wide (no auto-install)
- Use ~M-.~ (xref-find-definitions) for go-to-definition
- Use ~M-,~ (xref-pop-marker-stack) to jump back
- Use ~M-?~ (xref-find-references) for references
- Works with built-in imenu and consult integration
