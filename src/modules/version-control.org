#+AUTHOR: Diacus Magnuz
#+STARTUP: overview

[[https://magit.vc/][Magit]] is a complete text based user interface to [[https://git-scm.com/][Git]].
#+begin_src emacs-lisp
  (use-package magit
    :ensure t
    :config
    (ep3c-define-key 'normal 'global
      "g g" 'magit
      "g f" 'magit-find-file
      "g F" 'magit-find-file-other-frame
      "g l" 'magit-log-current
      "g b" 'magit-log-all-branches)
    (setq
     magit-log-arguments '("-n256" "--graph" "--decorate" "--color")
     magit-diff-refine-hunk t))
#+end_src
