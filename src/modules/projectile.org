#+TITLE: Projectile integration library settings
#+AUTHOR: Diacus Magnuz
#+STARTUP: overview

[[https://docs.projectile.mx/projectile/index.html][Projectile]] is a project integration library for Emacs. Its goal is to
provide features operating on a project level without introducing
external dependencies (when feasable).

#+begin_src emacs-lisp
  (use-package projectile
    :ensure t
    :config
    (ep3c-define-key 'normal 'global
      "p p" 'projectile-switch-project
      "p k" 'projectile-kill-buffers
      "p f" 'projectile-find-file
      "p F" 'projectile-find-file-other-frame
      "p b" 'projectile-switch-to-buffer
      "p B" 'projectile-switch-to-buffer-other-frame
      "p c" 'projectile-compile-project
      "p t" 'projectile-test-project)
    (projectile-mode +1)
    ;; Projectile ships its own C-c p prefix in `projectile-mode-map';
    ;; being a minor mode, it would shadow the non-evil leader key,
    ;; which owns the `p' prefix for the bindings above.
    (define-key projectile-mode-map (kbd "C-c p") nil)
    ;; Use default completion instead of ivy, since ivy is not installed.
    ;; This prevents "Cannot load projectile" error during startup.
    (setq projectile-completion-system 'default))

  ;; Run prefix (SPC p r): starts different REPLs in the project root.
  ;; Lowercase opens the REPL in a popup window, uppercase selects it in
  ;; the current window and reuses the project-specific buffer, like the
  ;; shell module does for SPC t.  The popup wrappers depend on
  ;; popup-window.el, installed by the shell module, so they wait for
  ;; it to load.  Note that `display-buffer' tries `display-buffer-alist'
  ;; before any explicit action the commands pass, hence even shell,
  ;; term and ielm, which request the same window, end up in the popup.
  (with-eval-after-load 'popup-window
    (defun projectile--popup-shell ()
      "Open the project shell in a popup window."
      (interactive)
      (funcall (make-repl-popup #'projectile-run-shell)))
    (defun projectile--popup-eshell ()
      "Open the project eshell in a popup window."
      (interactive)
      (funcall (make-repl-popup #'projectile-run-eshell)))
    (defun projectile--popup-term ()
      "Open the project term in a popup window."
      (interactive)
      (funcall (make-repl-popup #'projectile-run-term)))
    (defun projectile--popup-ielm ()
      "Open the project ielm in a popup window."
      (interactive)
      (funcall (make-repl-popup #'projectile-run-ielm)))
    (defun projectile--popup-vterm ()
      "Open the project vterm in a popup window."
      (interactive)
      (funcall (make-repl-popup #'projectile-run-vterm)))
    ;; Projectile has no projectile-run-ghostel of its own: ghostel
    ;; opens its terminal in `default-directory' like the other REPL
    ;; commands, so this is their shared shape.
    (defun projectile-run-ghostel ()
      "Open the project ghostel terminal in the current window."
      (interactive)
      (let ((default-directory (projectile-project-root)))
        (ghostel)))
    (defun projectile--popup-ghostel ()
      "Open the project ghostel terminal in a popup window."
      (interactive)
      (funcall (make-repl-popup #'projectile-run-ghostel)))
    (ep3c-define-key 'normal 'global
      "p r s" #'projectile--popup-shell
      "p r S" #'projectile-run-shell
      "p r e" #'projectile--popup-eshell
      "p r E" #'projectile-run-eshell
      "p r t" #'projectile--popup-term
      "p r T" #'projectile-run-term
      "p r g" #'projectile--popup-ghostel
      "p r G" #'projectile-run-ghostel
      "p r i" #'projectile--popup-ielm
      "p r I" #'projectile-run-ielm)
    ;; vterm is UNIX only: the v/V pair waits for vterm to load.
    (unless (member system-type '(ms-dos windows-nt cygwin))
      (with-eval-after-load 'vterm
        (ep3c-define-key 'normal 'global
          "p r v" #'projectile--popup-vterm
          "p r V" #'projectile-run-vterm))))
#+end_src
